# Plan View Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let the pane show the Markdown plan a session is following: all phases as one line each and one focused phase opened up, switched by a `[ Plan ]` / `[ Todo ]` button and `‹ ›` buttons.

**Architecture:** Two pure modules (`plan.ts`: parse, done signals, which file is the plan; `planview.ts`: lines for the phase list and the focused phase) and wiring in `register.tsx` (two atoms, a capture of the plan path in the existing `tool.call` hook, a 5 s file read, a `ui.press` hook, a command, a toast, the view). Types in `types/index.d.ts`.

**Tech Stack:** TypeScript on the Claude Code mod API; `claude plugin test`, `claude plugin validate`.

**Spec:** `docs/superpowers/specs/2026-10-16-plan-view-design.md` (task 1 records one refinement of its parsing rule: a horizontal rule ends a `##` section).

- Work in `~/claude-mods/task-progress`. Source style as the existing hooks: no semicolons, single quotes, 2-space indent.

- Strings from files are stripped of control characters and cut to 60 characters before they are drawn.

- Existing tests (145 at the start) stay green after every task. `hooks/register.tsx` stays under 800 lines.

## Review Focus

1. **Our own plans** (tasks as `###` under no phase heading, boxes unchecked, progress carried by `Task N:` todo labels) must read as one phase with the right done counts (Task 1, Task 3).
2. **Odd files:** CRLF, fenced code containing `- [ ]`, a `---` in the middle, a file with no task, 500+ tasks, huge file, control characters in headings (Task 1).
3. **Tiny panes and 1-phase or 12-phase plans** must not overflow or hide the focused phase (Task 2, Task 3).
4. **A plan file that vanishes, is unreadable or is rewritten mid-read** keeps the last plan and never disturbs the session (Task 3).
5. **Focus drift:** `‹ ›` at the ends, focus after 60 s, a plan that shrinks below the focus index (Task 2, Task 4).
6. **Toast storms:** a phase announces once; phases already complete when a plan is first read never announce; re-reading the same file never re-announces (Task 4).

import type { Phase, Plan, PlanTask } from '../types'

/** A plan file larger than this is not read. */
export const MAX_BYTES = 1_000_000
const MAX_TASKS = 500
const MAX_PHASES = 12
const NAME_MAX = 60
const CONTROL_CHARS = /[\u0000-\u001f\u007f-\u009f  ]/g
const HAS_CONTROL_CHARS = /[\u0000-\u001f\u007f-\u009f  ]/
const HEADING = /^(#{1,6})\s+(.*?)\s*#*\s*$/
const BOX = /^\s*(?:[-*+]|\d+\.)\s+\[([ xX])\]\s*(.*)$/
const RULE = /^\s*([-*_])(\s*\1){2,}\s*$/
const FENCE = /^\s*(```|~~~)/
const PLAN_TOOLS: readonly string[] = ['Read', 'Edit', 'Write', 'MultiEdit']

const clean = (text: string): string => text.replace(CONTROL_CHARS, ' ').replace(/\s+/g, ' ').trim().slice(0, NAME_MAX)

const toTask = (name: string, done: number, total: number): PlanTask => {
  const match = /^task\s+(\d+)\b/i.exec(name)

  return { name, number: match?.[1] === undefined ? null : Number(match[1]), done, total, isDone: total > 0 && done === total }
}

/**
 * The phases and tasks of a Markdown plan, or undefined when it has no task. A task is a `###` heading with checkboxes under it;
 * a phase is the `##` heading above its tasks (a horizontal rule ends that section, so tasks after one belong to a phase named
 * after the plan); boxes directly under a `##` heading are one-line tasks. Boxes in code fences are ignored.
 */
export function parsePlan(text: string, fileName: string): Plan | undefined {
  let title = ''
  let section: string | undefined
  let phase: Phase | undefined
  let open: { name: string; done: number; total: number } | undefined
  let taskCount = 0
  let fence: string | undefined
  const phases: Phase[] = []

  const phaseFor = (): Phase | undefined => {
    if (phase !== undefined) return phase
    if (phases.length >= MAX_PHASES) return undefined
    phase = { name: section ?? (clean(title) || fileName), tasks: [] }
    phases.push(phase)

    return phase
  }
  const add = (name: string, done: number, total: number): void => {
    if (taskCount >= MAX_TASKS) return
    const target = phaseFor()
    if (target === undefined) return
    target.tasks.push(toTask(name, done, total))
    taskCount += 1
  }
  const flush = (): void => {
    if (open !== undefined && open.total > 0) add(open.name, open.done, open.total)
    open = undefined
  }

  for (const line of text.split(/\r?\n/)) {
    const fenced = FENCE.exec(line)
    if (fenced !== null) {
      fence = fence === undefined ? fenced[1] : fence === fenced[1] ? undefined : fence
      continue
    }
    if (fence !== undefined) continue
    if (RULE.test(line)) {
      flush()
      section = undefined
      phase = undefined
      continue
    }
    const heading = HEADING.exec(line)
    if (heading !== null) {
      const level = heading[1]?.length ?? 0
      const name = clean(heading[2] ?? '')
      if (level <= 3) flush()
      if (level === 1) {
        title = title === '' ? name : title
        section = undefined
        phase = undefined
      } else if (level === 2) {
        section = name
        phase = undefined
      } else if (level === 3) open = { name, done: 0, total: 0 }
      continue
    }
    const box = BOX.exec(line)
    if (box === null) continue
    const isChecked = box[1] !== ' '
    if (open !== undefined) {
      open.total += 1
      if (isChecked) open.done += 1
    } else if (section !== undefined) add(clean(box[2] ?? ''), isChecked ? 1 : 0, 1)
  }
  flush()
  const kept = phases.filter(p => p.tasks.length > 0)

  return kept.length === 0 ? undefined : { title: clean(title) || fileName, phases: kept }
}

/** Marks as done the tasks whose number appears in a completed todo label starting `Task N`. */
export function applyDone(plan: Plan, completedLabels: readonly string[]): Plan {
  const numbers = new Set<number>()
  for (const label of completedLabels) {
    const match = /^\s*Task\s*(\d+)\b/i.exec(label)
    if (match?.[1] !== undefined) numbers.add(Number(match[1]))
  }
  if (numbers.size === 0) return plan

  return {
    ...plan,
    phases: plan.phases.map(p => ({ ...p, tasks: p.tasks.map(t => (t.number !== null && numbers.has(t.number) ? { ...t, isDone: true } : t)) })),
  }
}

export const phaseDoneCount = (phase: Phase): number => phase.tasks.filter(t => t.isDone).length

/** The first phase with an unfinished task; the last phase when everything is done. */
export function currentPhase(plan: Plan): number {
  const index = plan.phases.findIndex(p => p.tasks.some(t => !t.isDone))

  return index === -1 ? plan.phases.length - 1 : index
}

/** The plan file a tool call names: an absolute `.md` path with a `plan` or `plans` directory, read, edited or written. */
export function planPathOf(tool: string, input: Record<string, unknown>): string | undefined {
  const path = input.file_path
  if (!PLAN_TOOLS.includes(tool) || typeof path !== 'string') return undefined
  if (!path.startsWith('/') || !path.endsWith('.md') || HAS_CONTROL_CHARS.test(path)) return undefined

  return /\/plans?\//.test(path) ? path : undefined
}

import type { Plan } from '../types'
import { phaseDoneCount } from './plan'
import { truncate } from './style'

export type Tone = 'text' | 'muted' | 'success' | 'active' | 'accent'
export type PlanLine = { text: string; tone: Tone; isBold: boolean }

/** The focus returns to the current phase this long after the last button press. */
export const FOCUS_RETURN_MS = 60_000
/** The phase list never takes more rows than this. */
export const MAX_LIST_ROWS = 8
const BAR_CELLS = 8
const CIRCLED = '①②③④⑤⑥⑦⑧⑨⑩⑪⑫'
const MIN_NAME = 4

export const phaseNumber = (index: number): string => CIRCLED[index] ?? String(index + 1)

const bar = (done: number, total: number): string => {
  const filled = total === 0 ? 0 : Math.round((done / total) * BAR_CELLS)

  return `${'▰'.repeat(filled)}${'▱'.repeat(BAR_CELLS - filled)}`
}

/** The phase the pane shows: the user's focus for a minute after a press (kept inside the plan), else the current phase. */
export function shownFocus(focus: number | null, pressedAt: number, now: number, current: number, count: number): number {
  const last = Math.max(0, count - 1)
  const base = focus !== null && now - pressedAt <= FOCUS_RETURN_MS ? focus : current

  return Math.min(last, Math.max(0, base))
}

/** One line per phase, scrolled so the focused phase shows; the focused and the current phase also get a bar. */
export function phaseLines(plan: Plan, focus: number, current: number, maxRows: number, width: number): PlanLine[] {
  const count = plan.phases.length
  const rows = Math.max(1, Math.min(maxRows, MAX_LIST_ROWS, count))
  const start = count <= rows ? 0 : Math.min(count - rows, Math.max(0, focus - Math.floor(rows / 2)))
  const shown = plan.phases.slice(start, start + rows).map((phase, offset) => {
    const index = start + offset
    const done = phaseDoneCount(phase)
    const total = phase.tasks.length
    const mark = done === total ? '✔' : index === current ? '▶' : '○'
    const withBar = index === focus || index === current
    const tail = ` ${mark} ${done}/${total}${withBar ? ` ${bar(done, total)}` : ''}`

    return { index, phase, mark, tail, done, total }
  })
  const room = Math.max(MIN_NAME, width - 2 - Math.max(...shown.map(s => s.tail.length)))
  const pad = Math.min(room, Math.max(...shown.map(s => s.phase.name.length)))

  return shown.map(s => {
    const name = truncate(s.phase.name, room).padEnd(pad)
    const tone: Tone = s.index === focus ? 'accent' : s.mark === '✔' ? 'success' : s.mark === '▶' ? 'active' : 'muted'

    return { text: `${phaseNumber(s.index)} ${name}${s.tail}`, tone, isBold: s.index === focus }
  })
}

/**
 * The tasks of one phase in `maxRows` rows: finished tasks before the current one collapse into `✔ N done` plus the last
 * finished one, the current task is marked, the next ones follow, and `… +N` counts the rest. A tiny budget drops the head first.
 */
export function taskLines(plan: Plan, focus: number, maxRows: number, width: number): PlanLine[] {
  const tasks = plan.phases[focus]?.tasks ?? []
  const rows = Math.max(1, maxRows)
  const room = Math.max(1, width - 2)
  const index = tasks.findIndex(t => !t.isDone)
  if (index === -1) return [{ text: `✔ all ${tasks.length} done`, tone: 'success', isBold: false }]

  const line = (mark: string, name: string, tone: Tone, isBold = false, extra = ''): PlanLine => ({
    text: `${mark} ${truncate(name, Math.max(1, room - extra.length))}${extra}`,
    tone,
    isBold,
  })
  const head: PlanLine[] = []
  if (index >= 2) head.push({ text: `✔ ${index - 1} done`, tone: 'muted', isBold: false })
  const before = tasks[index - 1]
  if (before !== undefined) head.push(line('✔', before.name, 'success'))
  const current = tasks[index]
  if (current === undefined) return head
  head.push(line('▶', current.name, 'active', true, current.total > 1 ? `  ${current.done}/${current.total}` : ''))
  if (head.length >= rows) return head.slice(head.length - rows)

  const rest = tasks.slice(index + 1)
  const remaining = rows - head.length
  const items = (list: typeof rest): PlanLine[] => list.map(t => (t.isDone ? line('✔', t.name, 'success') : line('○', t.name, 'text')))
  if (rest.length <= remaining) return [...head, ...items(rest)]

  return [...head, ...items(rest.slice(0, remaining - 1)), { text: `… +${rest.length - (remaining - 1)}`, tone: 'muted', isBold: false }]
}
